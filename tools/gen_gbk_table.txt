# -*- coding: utf-8 -*-
# 生成 src/core/gbk_table.h：从本机 CPython gbk codec 逐对 strict 试解提取映射表。
# 由 tools\runcross.bat / 开发者手动运行：python tools\gen_gbk_table.txt
import os

OUT = os.path.join(os.path.dirname(os.path.abspath(__file__)),
                   "..", "src", "core", "gbk_table.h")

entries = []
for lead in range(0x81, 0xFF):
    trails = list(range(0x40, 0x7F)) + list(range(0x80, 0x100))
    for trail in trails:
        try:
            ch = bytes([lead, trail]).decode("gbk", "strict")
            cp = ord(ch[0])
        except Exception:
            cp = 0
        if cp:
            li = lead - 0x81
            ti = trail - 0x40 if trail < 0x7F else trail - 0x41
            entries.append((li, ti, cp))

entries.sort(key=lambda e: (e[0], e[1]))
with open(OUT, "w", encoding="ascii") as f:
    f.write("// AUTO-GENERATED from CPython gbk codec (strict per-pair probe). DO NOT EDIT.\n")
    f.write("// lead 0x81-0xFE x trail 0x40-0x7E,0x80-0xFE; value 0 = invalid pair\n")
    f.write("#pragma once\n#include <cstdint>\nnamespace sncom {\n")
    f.write("struct GbkEntry { uint8_t li, ti; uint16_t cp; };\n")
    f.write("inline const GbkEntry kGbkTable[] = {\n")
    line = ""
    for i, (li, ti, cp) in enumerate(entries):
        line += "{%d,%d,0x%04X}," % (li, ti, cp)
        if i % 8 == 7:
            f.write(line + "\n")
            line = ""
    if line:
        f.write(line + "\n")
    f.write("};\n")
    f.write("inline uint16_t GbkLookup(uint8_t lead, uint8_t trail) {\n")
    f.write("    if (lead < 0x81 || lead > 0xFE) return 0;\n")
    f.write("    if (trail < 0x40 || trail == 0x7F || trail > 0xFE) return 0;\n")
    f.write("    uint8_t li = (uint8_t)(lead - 0x81);\n")
    f.write("    uint8_t ti = (uint8_t)(trail < 0x7F ? trail - 0x40 : trail - 0x41);\n")
    f.write("    uint16_t q = (uint16_t)(li * 256 + ti);\n")
    f.write("    int lo = 0, hi = (int)(sizeof(kGbkTable) / sizeof(kGbkTable[0]));\n")
    f.write("    while (lo < hi) {\n")
    f.write("        int mid = (lo + hi) / 2;\n")
    f.write("        uint16_t a = (uint16_t)(kGbkTable[mid].li * 256 + kGbkTable[mid].ti);\n")
    f.write("        if (a < q) lo = mid + 1; else hi = mid;\n")
    f.write("    }\n")
    f.write("    if (lo < (int)(sizeof(kGbkTable) / sizeof(kGbkTable[0])) &&\n")
    f.write("        kGbkTable[lo].li == li && kGbkTable[lo].ti == ti) return kGbkTable[lo].cp;\n")
    f.write("    return 0;\n")
    f.write("}\n")
    f.write("} // namespace sncom\n")
print("entries:", len(entries), "->", os.path.abspath(OUT))
